cmake_minimum_required(VERSION 3.20)
project(tangxin_media VERSION 0.5.0 LANGUAGES CXX)
add_executable(tangxin-media src/merge.cpp)
target_compile_features(tangxin-media PRIVATE cxx_std_17)
set_target_properties(tangxin-media PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(tangxin-media PRIVATE /W4 /utf-8)
else()
  target_compile_options(tangxin-media PRIVATE -Wall -Wextra -Wpedantic)
  if(WIN32)
    target_link_options(tangxin-media PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME merge-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_merge.py $<TARGET_FILE:tangxin-media>)
